router.post('/signup',
[
body('name').notEmpty().withMessage('name reuired'),
body('email').isEmpty().withMessage('give correct email'),
body('password').isLength({min:6}).withMessage('minimum password 6 required')
],
async(req,res,next)=>{
const errors = validationResult(req)
if(!errors.isEmpty()){
return res.status(400).json({ success: false, errors: errors.array() });
}

const {name,email,password} = req.body

const isUserExsist = await User.findOne({email});
if(isUserExsist ){
return res.status(400).json({sucess: false, message: 'user exsists'})
}

const user = await User.create({name,email,password})
const token = generateToken(user.id)

res.status(201).json({
success: true,
message: 'user created',
token: token,
user:{
id: user.id,
name: user.name,
email: user.email
}
})
})
